STEP 1 — Push dari mesin development (Windows) — OTOMATIS
================================================================================
Jalankan SATU perintah ini di PowerShell (folder harus punya .git + remote):

  cd D:\Project\Klip
  Set-ExecutionPolicy -Scope Process Bypass -Force
  .\docs\scripts\push-to-sit.ps1

Atau Git Bash:

  cd /d/Project/Klip
  bash docs/scripts/push-to-sit.sh

Manual (jika script gagal):

  git checkout SIT
  git pull origin SIT
  git add -u backend/src frontend/src docs/scripts
  git commit -m "feat(shipping-perf): summary cards, vessel modal, table UI, SIT deploy"
  git push origin SIT

Contoh commit message (release Shipping Performance + modal):

  feat(shipping-perf): asymmetrical summary cards, vessel history modal, table UI sync

Catatan:
  - Push STEP 1 harus selesai sebelum STEP 2 di server.
  - Jika push ditolak: git pull --rebase origin SIT && git push origin SIT
  - Jika ERROR "folder .git tidak ada": buka folder clone Git yang benar (bukan copy tanpa .git).

padahal local sudah CANCELLED (duplikat PO; keeper = OP-LAND-160720260073).

cleanup belum dijalankan di DB SIT. Migration 119 TIDAK membatalkan op ini.

----- A) PuTTY → Backend 172.28.92.57 — pastikan code + migrate -----

  cd /opt/klip
  git fetch origin
  git checkout SIT
  git pull origin SIT
  git log -1 --oneline
  # Harus mengandung commit po-identity (mis. 5f9d3aa) atau lebih baru

  docker compose -f docker-compose.backend.yml up -d --build
  docker compose -f docker-compose.backend.yml ps
  curl -s http://127.0.0.1:5001/health

  # Cek migrate 119 sudah masuk schema_migrations
  docker compose -f docker-compose.backend.yml exec -T postgres \
    psql -U "${DB_USER:-postgres}" -d "${DB_NAME:-klip_db}" -c \
    "SELECT filename, applied_at FROM schema_migrations
     WHERE filename = '119_po_primary_identity.sql';"

  # Jika 0 baris → jalankan migrate manual, lalu ulangi SELECT di atas
  docker compose -f docker-compose.backend.yml exec backend \
    node dist/database/migrate.js

  # Pastikan script compiled ada di image
  docker compose -f docker-compose.backend.yml exec backend \
    ls -la dist/scripts/cleanupDuplicateTruckingByPo.js \
           dist/scripts/refreshTruckingPipelineSummary.js

----- B) BEFORE — bukti 1 baris masih aktif (In Progress) -----

  docker compose -f docker-compose.backend.yml exec -T postgres \
    psql -U "${DB_USER:-postgres}" -d "${DB_NAME:-klip_db}" -c "
  SELECT t.operation_id, t.status,
         (SELECT COUNT(DISTINCT da.progress_date)
            FROM trucking_daily_actuals da
           WHERE da.trucking_operation_id = t.id) AS wb_dates
  FROM trucking_operations t
  JOIN contracts c ON c.id = t.contract_id
  WHERE TRIM(c.po_number::text) = '1001031094'
  ORDER BY t.operation_id;
  "

  # Harapan BEFORE: dua baris aktif ATAU satu IN_PROGRESS untuk 300620260835
  # (status DB bisa IN_PROGRESS / PLANNED; UI menampilkan In Progress)

----- E) Manual refresh pipeline -----

----- F) AFTER — verifikasi DB -----

  docker compose -f docker-compose.backend.yml exec -T postgres \
    psql -U "${DB_USER:-postgres}" -d "${DB_NAME:-klip_db}" -c "
  SELECT t.operation_id, t.status,
         (SELECT COUNT(DISTINCT da.progress_date)
            FROM trucking_daily_actuals da
           WHERE da.trucking_operation_id = t.id) AS wb_dates
  FROM trucking_operations t
  JOIN contracts c ON c.id = t.contract_id
  WHERE TRIM(c.po_number::text) = '1001031094'
  ORDER BY CASE WHEN UPPER(t.status)='CANCELLED' THEN 1 ELSE 0 END, t.operation_id;
  "

  # Harapan AFTER:
  #   OP-LAND-160720260073 | COMPLETED | wb_dates > 0
  #   OP-LAND-300620260835 | CANCELLED | wb_dates = 0

    psql -U "${DB_USER:-postgres}" -d "${DB_NAME:-klip_db}" -c "
  SELECT stage FROM trucking_list_stage_snapshot sn
  JOIN trucking_operations t ON t.id = sn.operation_id
  WHERE t.operation_id = 'OP-LAND-300620260835';
  "

  # Harapan snapshot stage = CANCELLED
