# SIT — repair parallel multi-STO collapse + materialize missing shipments
# Case: PO 1581000931 — Shipments showed 4928+4929 only; Contracts showed 4927+4928+4929.
# Root cause on RDS: shipment_id=1586004929 had operation_id=1586004927 (false match),
# so backfill skipped creating 1586004927.
#
# Prereq: pull latest SIT + rebuild backend.

cd /opt/klip
git fetch origin
git checkout SIT
git pull origin SIT
docker compose -f docker-compose.backend.yml up -d --build

# Confirm DB is Aliyun RDS (not co-located postgres)
docker exec klip-backend printenv DB_HOST DB_NAME NODE_ENV

# Diagnose (shows collisions + gaps)
docker exec -w /app klip-backend node dist/scripts/backfillMissingSapStoShipments.js --diagnose --po 1581000931

# Dry-run: lists operation_id clears + missing STO creates
docker exec -w /app klip-backend node dist/scripts/backfillMissingSapStoShipments.js --dry-run --po 1581000931

# Repair: clear colliding operation_id, then create missing shipment rows
docker exec -w /app klip-backend node dist/scripts/backfillMissingSapStoShipments.js --repair-operation-id --po 1581000931

# Verify via API DB
docker exec -i -w /app klip-backend node <<'EOF'
const pool = require('./dist/database/connection').default;
(async () => {
  const c = await pool.connect();
  try {
    const ships = await c.query(
      `select s.shipment_id, s.operation_id, s.status
       from shipments s
       join contracts ct on ct.id = s.contract_id
       where trim(ct.po_number::text) = '1581000931'
       order by 1`
    );
    console.log('ships', ships.rows);
  } finally {
    c.release();
    process.exit(0);
  }
})().catch((e) => { console.error(e); process.exit(1); });
EOF
